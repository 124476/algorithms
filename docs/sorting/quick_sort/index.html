// Quick sort — O(n log n) в среднем, O(n²) в худшем.
// Выбирает опорный элемент (pivot), разбивает массив на две части:
// меньше pivot и больше pivot, затем рекурсивно сортирует обе.

int partition(vector<int>& a, int l, int r) {
    int pivot = a[r];
    int i = l - 1;

    for (int j = l; j < r; ++j) {
        if (a[j] <= pivot) {
            ++i;
            swap(a[i], a[j]);
        }
    }

    swap(a[i + 1], a[r]);
    return i + 1;
}

void quickSort(vector<int>& a, int l, int r) {
    if (l >= r) return;

    int p = partition(a, l, r);
    quickSort(a, l, p - 1);
    quickSort(a, p + 1, r);
}
