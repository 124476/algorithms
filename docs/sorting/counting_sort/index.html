counting sort

// Counting sort — O(n + k), где k — диапазон значений.
// Работает только для целых неотрицательных (или со сдвигом).
// Не сравнивает элементы — считает количество каждого значения.

vector<int> countingSort(const vector<int>& a) {
    if (a.empty()) return {};

    int mx = *max_element(a.begin(), a.end());
    vector<int> count(mx + 1, 0);

    for (int x : a) ++count[x];

    vector<int> res;
    res.reserve(a.size());

    for (int v = 0; v <= mx; ++v) {
        while (count[v]-- > 0) {
            res.push_back(v);
        }
    }

    return res;
}
