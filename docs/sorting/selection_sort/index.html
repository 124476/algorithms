// Selection sort — O(n²).
// На каждом шаге находит минимум в неотсортированной части
// и ставит его в начало этой части.

void selectionSort(vector<int>& a) {
    int n = (int)a.size();

    for (int i = 0; i < n - 1; ++i) {
        int minIdx = i;

        for (int j = i + 1; j < n; ++j) {
            if (a[j] < a[minIdx]) {
                minIdx = j;
            }
        }

        if (minIdx != i) swap(a[i], a[minIdx]);
    }
}
