insertion sort

// Insertion sort — O(n²), но O(n) на почти отсортированных данных.
// Строит отсортированный массив по одному элементу за раз,
// вставляя каждый новый элемент на его место.

void insertionSort(vector<int>& a) {
    int n = (int)a.size();

    for (int i = 1; i < n; ++i) {
        int key = a[i];
        int j = i - 1;

        while (j >= 0 && a[j] > key) {
            a[j + 1] = a[j];
            --j;
        }

        a[j + 1] = key;
    }
}
